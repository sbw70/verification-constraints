+static bool send_client_ready(int client_sock)
+{
+    static const char ready[] =
+        "{\"ready\":true,\"test\":\"ESP_LOCAL_007\"}\n";
+
+    size_t sent_total = 0U;
+    const size_t ready_length = sizeof(ready) - 1U;
+
+    while (sent_total < ready_length) {
+        ssize_t sent = send(
+            client_sock,
+            ready + sent_total,
+            ready_length - sent_total,
+            0
+        );
+
+        if (sent <= 0) {
+            return false;
+        }
+
+        sent_total += (size_t)sent;
+    }
+
+    return true;
+}
